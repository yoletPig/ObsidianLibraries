### 1.1 raft_node.py：Raft 核心（选举 + 日志复制）

```python
# raft_node.py —— 单文件实现 Raft 核心：选举、日志复制、提交、KV 状态机
import asyncio, random

ELECTION_MIN, ELECTION_MAX = 0.3, 0.6     # 选举超时区间（秒）
HEARTBEAT = 0.1                            # 心跳间隔

class RaftNode:
    def __init__(self, name, peers, net):
        self.name, self.peers, self.net = name, peers, net  # net: 网络层(见 kv_server)
        # ---- 持久化状态（真实系统要落盘，这里存内存）----
        self.term = 0                      # 当前任期
        self.voted_for = None              # 本任期投给谁
        self.log = []                      # [(term, cmd)] cmd: ("set",k,v)/("del",k)
        # ---- 易失状态 ----
        self.role = "follower"
        self.commit_index = -1
        self.last_applied = -1
        self.next_index = {p: 0 for p in peers}    # 每个 follower 下一条要发的 index
        self.state = {}                    # 应用层 KV 状态机
        self.deadline = 0.0
        self._task = asyncio.create_task(self._tick())
        self._waiters = []                 # 等待"某 index 提交"的 future 队列

    # ============ 时钟循环：心跳与选举 ============
    async def _tick(self):
        while True:
            await asyncio.sleep(0.01)
            now = asyncio.get_event_loop().time()
            if self.role == "leader":
                await self._broadcast_append()          # leader 持续心跳/复制
                await asyncio.sleep(HEARTBEAT)
            elif now >= self.deadline:
                await self._start_election()

    def _reset_deadline(self):
        self.deadline = asyncio.get_event_loop().time() + random.uniform(ELECTION_MIN, ELECTION_MAX)

    # ============ 选举 ============
    async def _start_election(self):
        self.term += 1
        self.role, self.voted_for = "candidate", self.name
        self._reset_deadline()
        my_last = (len(self.log) - 1, self.log[-1][0] if self.log else 0)
        votes = {self.name}
        replies = [self.net.rpc(self.name, p, "vote", {"term": self.term, "idx": my_last[0],
                                                      "lt": my_last[1], "cand": self.name})
                   for p in self.peers]
        for fut in asyncio.as_completed(replies):
            r = await fut
            if r and r.get("ok"):
                votes.add(r["from"])
            if len(votes) > (len(self.peers) + 1) // 2:
                self.role = "leader"                       # 多数派 → 当选
                for p in self.peers: self.next_index[p] = len(self.log)
                print(f"[{self.name}] 当选 leader term={self.term}")
                await self._broadcast_append()
                return

    # ============ 日志复制 ============
    async def _broadcast_append(self):
        for p in self.peers:
            ni = self.next_index[p]
            prev = (ni - 1, self.log[ni - 1][0] if ni > 0 else 0)
            entries = self.log[ni:]
            asyncio.create_task(self._append_to(p, prev, entries))

    async def _append_to(self, p, prev, entries):
        r = await self.net.rpc(self.name, p, "append", {"term": self.term, "prev_idx": prev[0],
                                                       "prev_term": prev[1],
                                                       "entries": entries, "commit": self.commit_index})
        if r and r.get("ok"):
            self.next_index[p] = prev[0] + len(entries) + 1   # 推进
            self._advance_commit()                            # 多数派 ack → 提交
        elif r and r.get("fail_idx") is not None:
            self.next_index[p] = max(0, r["fail_idx"])        # 一致性检查失败 → 回溯

    def _advance_commit(self):
        """提交规则：找到'多数派已复制 且 是本任期'的最大 index"""
        n = len(self.peers) + 1
        for i in range(len(self.log) - 1, self.commit_index, -1):
            if self.log[i][0] != self.term:                   # 只直接提交本任期条目(5.4.2)
                continue
            acks = 1 + sum(1 for p in self.peers if self.next_index[p] > i)
            if acks > n // 2:
                self.commit_index = i
                self._apply()
                for w in [f for f in self._waiters if f["idx"] == i]:
                    w["fut"].set_result(True)
                self._waiters = [f for f in self._waiters if f["idx"] != i]
                return

    def _apply(self):
        while self.last_applied < self.commit_index:
            self.last_applied += 1
            _, cmd = self.log[self.last_applied]
            if cmd[0] == "set":  self.state[cmd[1]] = cmd[2]
            elif cmd[0] == "del": self.state.pop(cmd[1], None)

    # ============ 客户端写：leader 追加并等待提交 ============
    async def propose(self, cmd):
        if self.role != "leader":
            return None                                       # follower 不接写(真实系统返回 leader 地址)
        self.log.append((self.term, cmd))
        fut = asyncio.get_running_loop().create_future()
        self._waiters.append({"idx": len(self.log) - 1, "fut": fut})
        self._advance_commit()
        try:
            await asyncio.wait_for(fut, timeout=1.0)          # 多数派 1s 内没确认 → 失败
            return True
        except asyncio.TimeoutError:
            return False

    # ============ RPC 处理（被网络层调用）============
    def handle_vote(self, m):
        if m["term"] > self.term:                             # 更大任期 → 更新、可投
            self.term, self.role, self.voted_for = m["term"], "follower", None
            self._reset_deadline()
        my_newer = (m["lt"] < (self.log[-1][0] if self.log else 0)) or \
                   (m["lt"] == (self.log[-1][0] if self.log else 0) and m["idx"] < len(self.log) - 1)
        if m["term"] == self.term and self.voted_for is None and not my_newer:
            self.voted_for = m["cand"]                        # 一票 + 日志检查(选举限制)
            self._reset_deadline()
            return {"ok": True, "from": self.name}
        return {"ok": False, "from": self.name}

    def handle_append(self, m):
        if m["term"] < self.term:                             # 旧 leader 的消息：拒绝(防脑裂)
            return {"ok": False, "from": self.name}
        self.role = "follower"; self._reset_deadline()        # 认可权威
        if m["term"] > self.term: self.term = m["term"]
        if m["prev_idx"] >= 0 and (m["prev_idx"] >= len(self.log) or
                                   self.log[m["prev_idx"]][0] != m["prev_term"]):
            return {"ok": False, "fail_idx": m["prev_idx"], "from": self.name}  # 日志匹配失败
        for i, e in enumerate(m["entries"]):                  # 截断冲突 + 追加
            idx = m["prev_idx"] + 1 + i
            if idx < len(self.log):
                if self.log[idx] != e: self.log = self.log[:idx]; self.log.append(e)
            else:
                self.log.append(e)
        if m["commit"] > self.commit_index:
            self.commit_index = min(m["commit"], len(self.log) - 1); self._apply()
        return {"ok": True, "from": self.name}
```

### 1.2 kv_server.py：网络层与集群启动

```python
# kv_server.py —— 用 asyncio.Queue 模拟网络，启动 3 节点 Raft 集群
import asyncio, random, sys
from raft_node import RaftNode

class SimNet:
    """极简网络层：按节点名投递 RPC；可注入丢包/分区/延迟。
    注意分区是双向的：src 或 to 任一在 partition 集合中，消息都到不了——
    只检查 to 是模拟网络时最常见的错误（被隔离的节点仍能外发）。"""
    def __init__(self, nodes):
        self.nodes, self.drop = nodes, 0.0
        self.partition = set()              # 集合内的节点与外界双向失联

    async def rpc(self, src: str, to: str, kind: str, msg: dict):
        if random.random() < self.drop or src in self.partition or to in self.partition:
            return None                     # 消息丢失（调用方按失败处理）
        await asyncio.sleep(random.uniform(0.001, 0.01))   # 模拟网络延迟
        node = self.nodes[to]
        return node.handle_vote(msg) if kind == "vote" else node.handle_append(msg)

async def main():
    random.seed(7)
    names = ["A", "B", "C"]
    nodes = {n: RaftNode(n, [], None) for n in names}
    net = SimNet(nodes)
    for n in nodes.values():    # 注意用 n.name（字符串）比较，别拿节点对象比——这是本文件曾埋过的真 bug
        n.peers = [x for x in names if x != n.name]; n.net = net
    await asyncio.sleep(0.8)                    # 等选举完成
    leader = next(n for n in nodes.values() if n.role == "leader")
    print("leader =", leader.name)
    # 正常写三条
    for kv in [("name", "zhang"), ("age", 25), ("city", "beijing")]:
        ok = await leader.propose(("set", *kv))
        print(f"set {kv} → {'已提交' if ok else '失败'}")
    # 制造分区：leader 被隔离（双向），写应失败；多数派侧选新 leader
    net.partition.add(leader.name)
    print(f"\n>>> 分区: {leader.name} 被隔离")
    await asyncio.sleep(0.8)                    # 等其余两节点超时并选出新 leader
    ok = await leader.propose(("set", "hack", "ghost"))    # 旧 leader 的幽灵写
    print(f"旧 leader 写入 → {'成功(有bug!)' if ok else '被多数派挡住（正确）'}")
    # 找新 leader 时必须排除旧 leader——它自己仍以为自己是 leader（本文件曾埋过的第二个真 bug）
    new_leader = next((n for n in nodes.values() if n.role == "leader" and n is not leader), None)
    if new_leader:
        ok = await new_leader.propose(("set", "city", "shanghai"))
        print(f"新 leader({new_leader.name}) 写入 → {'已提交' if ok else '失败'}")
    net.partition.clear()                       # 分区恢复：旧 leader 收到更高任期心跳 → 回归 follower
    await asyncio.sleep(0.3)
    if new_leader:                              # 新 leader 再写一条：覆盖旧 leader 的未提交幽灵日志
        ok = await new_leader.propose(("set", "zone", "pd"))
        print(f"恢复后写入 → {'已提交' if ok else '失败'}（旧 leader 的幽灵日志被截断替换）")
    await asyncio.sleep(0.5)
    print("\n最终三节点状态一致检查:")
    for n in nodes.values():
        print(f"  {n.name}: {n.state} commit={n.commit_index} 日志条数={len(n.log)}")

if __name__ == "__main__":
    asyncio.run(main())
```

### 1.3 client.py：线性一致读的实现

```python
# client.py —— 读请求走 leader 且确认自己仍是 leader（简化 ReadIndex 思想）
import asyncio, random
from raft_node import ELECTION_MIN, ELECTION_MAX

async def linearizable_read(cluster, key):
    """向'自认为 leader'的节点读；读前用一次多数派心跳确认领导权未过期（ReadIndex 简化版）。
    真实系统：leader 把'读时刻的 commit_index'记下，等 apply 追上再读本地状态机。"""
    for node in cluster:
        if node.role == "leader":
            # 关键：先与多数派交换一次心跳确认领导权（防读到已过期的旧 leader）
            acks = 1 + sum(1 for p in node.peers
                           if await node.net.rpc(node.name, p, "append",
                                                 {"term": node.term, "prev_idx": -1,
                                                  "prev_term": 0, "entries": [],
                                                  "commit": node.commit_index}))
            if acks > len(cluster) // 2:
                return node.state.get(key)      # 确认仍是多数派认可的 leader → 本地读即线性一致
    return None

async def main():
    random.seed(7)
    names = ["A", "B", "C"]
    nodes = {n: RaftNode(n, [], None) for n in names}

    # （完整串联运行见练习 1：把 SimNet 移入公共模块后在此启动集群并调用 linearizable_read）

if __name__ == "__main__":
```

代码与理论的对照表（复习用）：

| 理论概念（第 05 章） | 代码落点 |
| --- | --- |
| 任期 = 逻辑时钟 | `self.term`，`handle_append/vote` 里 `term < self.term` 即拒绝 |
| 选举限制 | `handle_vote` 的 `my_newer` 日志比较 |
| 一票限制 | `self.voted_for` 单值，本任期只写一次 |
| 日志匹配 | `handle_append` 的 `prev_idx/prev_term` 检查 + 截断 |
| 提交规则（本任期 + 多数派） | `_advance_commit` 的两个 if |
| 脑裂防护 | 分区中旧 leader `propose` 永远凑不齐多数派 → 超时失败 |
| 线性一致读 | client.py 的确认心跳 + 本地读 |

## 2. 面试高频 30 题（带答案要点）

**理论篇**

1. **CAP 定理到底说了什么？** 分区发生时线性一致（C）与可用性（A）不可兼得；P 不可选；取舍是操作级不是系统级标签。别忘补 PACELC。
2. **为什么分布式不能靠物理时钟排序事件？** 漂移 + NTP 阶跃 + 跨机无共同现在；"先发生"需要 happens-before 定义；Lamport 时钟保因果方向，向量时钟可判并发。
3. **两将军问题与拜占庭问题的区别？** 前者信道不可靠（丢消息）下无法 100% 达成一致；后者节点还会说谎，容 f 个叛徒需 3f+1；数据中心只需崩溃容错 2f+1。
4. **FLP 不可能定理说了什么，工程怎么绕？** 纯异步 + 1 崩溃故障则确定性共识无法保证终止；工程用超时/随机化把系统变成部分同步，牺牲最坏情况终止性。
5. **Raft 为什么必须多数派提交？** 两个"多数派"必相交 + 一票限制 → 同一任期不可能两个 leader；相交保证已提交条目存在于任何未来 leader → 已提交不被覆盖。少数派提交会造成日志分叉不可修复。
6. **Raft 如何防止脑裂？** 任期比较拒绝旧 leader 消息 + 旧 leader 凑不齐多数派无法提交 + 随机选举超时防选票瓜分。旧分区侧一切写入都不会对客户端返回成功。
7. **Raft 的选举限制是什么，为什么需要？** 投票前比较候选人日志（先比最后条目 term 再比 index），不比我新就不投——保证新 leader 拥有全部已提交条目（Leader 完备性）。
8. **为什么 Raft leader 不直接提交前任任期的条目？** 旧条目即使过半复制，也不能证明"不会有更新的 leader 用冲突条目覆盖它"；只有提交本任期条目才能间接确认旧条目安全（5.4.2 节）。
9. **Quorum W+R>N 保证了什么、不保证什么？** 保证读到至少一份最新副本；不保证线性一致（并发交错/ sibling 合并仍可能给出旧序）。
10. **线性一致读在 Raft 上怎么做？** ReadIndex（读前确认领导权 + 等应用追上 commit）或 Lease Read（租约期内免确认，依赖时钟假设）。直接读 leader 状态机是漏洞。
11. **最终一致的系统怎么实现"读己之写"？** 会话粘滞（写后短窗内读主/同副本）或客户端带时间戳门槛读。不需要全局强一致。
12. **幂等为什么是必须的？** at-least-once 下重投不可避免（发送超时重发、ack 丢失重投）；"处理与确认跨系统"决定 exactly-once 只能端到端达成 → 让重复无害。
13. **分布式锁的正确姿势？** TTL 判断不可靠（时钟 + 进程暂停）；要 fencing token（单调纪元）由资源方拒绝旧纪元；纪元单调性由共识保证。
14. **2PC 的致命弱点？** 协调者在"落盘决定后、通知完成前"宕机 → 所有参与者卡在 in-doubt 持锁阻塞。3PC 想修却依赖网络延迟上界假设，分区下破坏原子性。
15. **Saga 的补偿与数据库回滚的区别？** 补偿是新业务操作（可见中间态、可能失败要重试、需要幂等）；隔离性缺失要业务防（状态机让中间态不可见）。

**工程篇**

16. **Kafka 为什么快？** 顺序写日志 + 零拷贝 sendfile + 批量压缩 + 分区并行（OS 09 章的 IO 直觉全用上）。
17. **消费堆积怎么办？** 查消费耗时瓶颈 → 扩消费者（≤ 分区数）→ 加分区（注意 key 顺序语义）→ 降级非关键消费；堆积是消费能力账单，不是 MQ bug。
18. **重试风暴怎么形成、怎么防？** 全体客户端同时重试叠加原始流量压垮刚恢复的下游；防：指数退避 + full jitter + 重试预算 + 重试只包一层（不级联）。
19. **熔断器三个状态与切换条件？** 关闭(统计失败率)→超阈值→打开(快速失败冷却期)→冷却到→半开(放探测)→成功关/失败开。阈值要带样本量防震荡。
20. **令牌桶 vs 漏桶？** 令牌桶限平均速率容忍突发；漏桶恒定流出绝对平滑。分布式限流要 Redis+Lua 原子或网关层实现。
21. **一致性哈希解决什么问题？** 取模扩容全量搬家 → 环上只迁 K/N；倾斜靠 vnode 大数定律（实测 vnodes=100 时极差 <1%）。
22. **为什么 Kafka 用固定分区数不用一致性哈希？** 分区是并行度与顺序的单位，固定数量让路由表小、副本放置可控；弹性扩分区成本高，所以按峰值预设。
23. **心跳超时怎么设？** 长于"正常间隔 × 抖动余量"且覆盖已知停顿（GC）；误判代价高于漏判，配合间接探测与滞回；Φ Accrual 用分布代替阈值。
24. **Gossip 为什么能收敛、多快？** 每轮每节点随机传 fanout 个同伴，未覆盖数指数衰减，O(log N) 轮；增量 + 版本号防旧消息回环。
25. **主从切换怎么防丢数据？** 半同步（至少一个从确认）或共识日志（Raft）；异步复制必丢"切换窗口内未同步"的已确认写——用"最大可用 or 最大安全"参数显式选择。

**场景篇**

26. **ZooKeeper 和 etcd 怎么选？** 都是小而强一致的协调存储：etcd（Raft、MVCC、watch、K8s 生态）、ZK（ZAB、临时节点、成熟稳定）；大量业务数据放谁都不对。
27. **Redis 主从 + 哨兵能保证不丢数据吗？** 不能——异步复制 + 哨兵判主窗口内已确认写可能丢；要强一致用 Redlock 也要配 fencing（且被 Kleppmann 批评），或换 Raft 系存储。
28. **分布式 ID 怎么生成？** UUID（无序索引差）、号段模式（DB 批量发号）、Snowflake（时间戳+机器+序列，趋势递增，本质混合时钟）；回拨要用等待或备用位处理。
29. **PD 分离解决什么？** prefill 计算密集 vs decode 访存密集，分开部署各自扩容，KV cache 跨节点传输衔接——y 轴拆分的 LLM 版。
30. **LLM 训练的 3D 并行各自解决什么？** DP=吞吐（梯度 AllReduce，2(N-1)/N 通信量）、TP=单层放不下（机内 NVLink、层内通信频繁）、PP=层数深（气泡 (p-1)/(m+p-1) 靠 micro-batch 稀释）。

## 3. 设计题答题框架

四步法（每道设计题都按这个骨架走，先总后分、先稳后优）：

1. **澄清与量化**：QPS/读写比/数据量/延迟要求/一致性要求/可用性目标。数字决定架构——1k QPS 和 100k QPS 的答案完全不同。
2. **主干设计**：先画最简单可行架构（无状态层 + 存储层 + 队列/缓存），再谈分区与复制。用本系列语言：x/y/z 轴 + 复制模式 + 一致性等级。
3. **深挖难点**：面试官追问的都是"故障时怎么办"——按三笔账（部分失败/不可靠网络/无全局时钟）逐项回答：超时？重试？幂等？补偿？
4. **权衡显式化**：每个选择说出放弃了什么。"我选异步复制买低延迟，代价是切换窗口可能丢已确认写，用半同步参数缓解"比"用 Redis 就行"高一个段位。

**秒杀系统**：漏斗分层——前端静态化+按钮防抖（挡 90%）、网关限流（令牌桶，按库存量发放请求额度）、Redis 预扣库存（Lua 原子 decr，防超卖）→ MQ 异步下单（削峰）→ DB 最终落库（唯一索引防重复下单）。一致性等级：库存扣减线性一致（Redis 单点原子或 Lua），订单落库最终一致（Outbox）。故障预案：Redis 挂→降级 DB 限流直购；MQ 堆积→消费者扩容；重复支付→幂等键（用户ID+场次ID）。

**短链系统**：发号器（Snowflake 或 DB 号段）→ Base62 编码 → KV 存储（哈希分区，key=短码）。302 重定向 + CDN 缓存热点。关键难点：防自定义短码冲突（唯一索引兜底）、统计点击数（异步累加 + 定期合并，容忍秒级误差）、过期回收（惰性删除 + 版本号）。容量账：10 亿短链 × 60B ≈ 60GB，单机都放得下，真要分布式只为吞吐与可用性——先说清这一点再画集群（考官在测你会不会"杀鸡用牛刀"）。

**Feed 流**：推模式（写扩散，发布时写进所有粉丝收件箱，读快写重，适合粉丝少）vs 拉模式（读扩散，读时合并关注者最新内容，写轻读重，适合大 V）vs 推拉结合（普通人推、大 V 拉——经验做法主流方案）。存储：收件箱用"用户ID 分区 + 时间戳倒序列表"（因果一致即可，第 03 章）；一致性要求：自己发的帖子自己立刻可见（读己之写，写后粘主）；热点用户发布风暴（千万粉丝写扩散爆炸）用"大 V 内容拉取 + 收件箱只存指针"化解。

## 常见误区

1. **"面试设计题要答出最先进的架构"**：考官要的是**取舍能力**不是名词堆砌。先给简单方案说出它的天花板，再按量化数字升级——"kv 单表 → 按用户分区 → 一致性哈希扩容"的演进叙事远胜上来就画 8 组件架构图。
2. **"手写 Raft 只是面试噱头，工作中用不上"**：读 etcd/TiKV/Raft 库源码时，选举/日志/快照的结构与你手写的完全同构（差异只在持久化、网络、性能工程）；这套骨架也是理解 K8s 控制面、TiDB Placement Rule 的门票。
3. **"30 题背完就行"**：面试官会从你的答案里挑一处追问两层（"多数派为什么防脑裂？"→"两个多数派为什么必相交？"→"如果节点数是偶数呢？"）。每题要能向下推一层，本系列各章的"为什么"小节就是为此准备的。
4. **"把模拟代码当生产代码"**：本篇 Raft 实现缺持久化、快照、成员变更、真实网络栈——它是教学骨架。生产永远用 etcd/raft 库（Go）或 alibaba/SOFAJRaft，不要自研共识（经验做法：自研共识是分布式领域最经典的"重新发明事故"来源）。
5. **"分布式面试只考理论"**：一半以上问题会落到"线上事故怎么排查"——堆积、超时抖动、切换丢数据、热点分区。答题时主动引用"我会先看什么指标、分几步定位"，把理论挂到排障动作上。

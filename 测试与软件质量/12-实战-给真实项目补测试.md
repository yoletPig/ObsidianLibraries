```python
# after —— src/agentutils/workspace.py
from pathlib import Path

class Workspace:
    """Agent 可见的文件沙箱：一切路径解析以 root 为锚。"""

    def __init__(self, root: Path):
        self._root = Path(root).resolve()          # resolve 折叠 .. 与符号链接

    def _resolve(self, path: str) -> Path:
        full = (self._root / path).resolve()
        if not str(full).startswith(str(self._root)):
            raise PermissionError(f"path escapes workspace: {path!r}")
        return full

    def read(self, path: str) -> str:
        return self._resolve(path).read_text(encoding="utf-8")

    def write(self, path: str, content: str) -> None:
        full = self._resolve(path)
        full.parent.mkdir(parents=True, exist_ok=True)
        full.write_text(content, encoding="utf-8")

    def list_dir(self, path: str = ".") -> list[str]:
        return sorted(p.name for p in self._resolve(path).iterdir())
```

```python
# after —— src/agentutils/adapters.py（薄壳真实现，只在集成层测）
import os
from .ports import NotesReader
from .workspace import Workspace

class WorkspaceNotes:
    def __init__(self, ws: Workspace, notes_name: str = "AGENT_NOTES.md"):
        self._ws, self._name = ws, notes_name
    def read(self, path: str) -> str:
        return self._ws.read(path if path == self._name else self._name)

class OpenAIClient:
    def __init__(self, base_url="https://api.openai.com/v1", api_key=None,
                 model="gpt-4o-mini", http_client=None):   # ⭐ 传输可注入（5 节的教训）
        import openai
        self._client = openai.OpenAI(
            base_url=base_url,
            api_key=api_key if api_key is not None else os.environ["AGENT_API_KEY"],
            http_client=http_client)
        self._model = model

    def chat(self, system: str, user: str) -> str:
        r = self._client.chat.completions.create(
            model=self._model,
            messages=[{"role": "system", "content": system},
                      {"role": "user", "content": user}])
        return r.choices[0].message.content or ""   # mypy 抓出 SDK 返回 str|None（实测）
```

```python
# after —— src/agentutils/agent.py（纯逻辑：依赖全注入）
from .ports import LLMClient, NotesReader

SYSTEM_PROMPT = "你是笔记助手，仅根据笔记回答，不知道的说不知道。"
MAX_PROMPT = 8000

class NotesAgent:
    def __init__(self, notes: NotesReader, llm: LLMClient):
        self._notes, self._llm = notes, llm

    def answer(self, question: str) -> str:
        ctx = self._notes.read("AGENT_NOTES.md")
        prefix, suffix = "笔记：\n", f"\n\n问题：{question}"
        budget = MAX_PROMPT - len(prefix) - len(suffix)   # 精确预算
        ctx = ctx[:budget]
        return self._llm.chat(SYSTEM_PROMPT, f"{prefix}{ctx}{suffix}")
```

**前后对比**（重构发生在 characterization 全绿保护下）：

| 维度 | before | after |
| --- | --- | --- |
| 根目录依赖 | 环境变量（每次读） | 构造函数注入 `Workspace(root)` |
| 逃逸校验 | 两处复制粘贴 | `_resolve` 单点 |
| LLM 依赖 | 函数内 `openai.OpenAI()` | `LLMClient` 端口 + `OpenAIClient` 适配器 |
| 可测性 | 必须环境变量 + 真 SDK | tmp_path + Fake 即可 |
| 行为变化 | — | **零**（characterization 移植后全绿） |

迁移期做了一次"characterization 退役"：旧函数式 API 的 5 条用例逐条改写为新类 API（`read_file("a.txt")` → `Workspace(root=tmp_path).read("a.txt")`），改写完的用例**升级为永久单元测试** `tests/unit/test_workspace.py`，原文件归档进 `docs-legacy/`。这就是 06 篇说的：characterization 是脚手架，不是永久房。

```python
# tests/unit/test_agent.py —— Fake 替身（04 篇：Fake 优于 Mock 的场景）
from agentutils.agent import NotesAgent, SYSTEM_PROMPT

class FakeNotes:
    def __init__(self, text): self._t = text
    def read(self, path): assert path == "AGENT_NOTES.md"; return self._t

class FakeLLM:
    def __init__(self, reply="ok"): self.calls = []; self._reply = reply
    def chat(self, system, user):
        self.calls.append((system, user))
        return self._reply

def test_prompt_contains_notes_and_question():
    llm = FakeLLM()
    NotesAgent(FakeNotes("会议纪要：周三发布"), llm).answer("何时发布？")
    system, user = llm.calls[0]
    assert system == SYSTEM_PROMPT and "何时发布？" in user

def test_truncates_long_notes():
    llm = FakeLLM()
    NotesAgent(FakeNotes("x" * 20_000), llm).answer("q")
    assert len(llm.calls[0][1]) <= 8000
```

`test_truncates_long_notes` 第一次运行就红了：第一版预算公式 `MAX_PROMPT - len(question) - 8` 少算了一个前缀字符，`len(user) == 8001`。**这就是测试的价值瞬间**：上下文超限在真实环境里表现为"偶发 API 报错"，定位要几小时；在这里是 0.1 秒的一行红字。修复后的精确预算版全绿。

## 5. 第四步：mock LLM 的边界——为什么不用 `patch("openai...")`

04 篇原则在本项目的两次落地：

1. **业务测试 mock 自家端口**：`FakeLLM` 是 `LLMClient` 端口的 Fake——记录调用、返回固定值，业务分支全覆盖（上面 3 条用例），从未出现 `patch("openai...")` 字样；
2. **不 mock 你不拥有的东西**：`OpenAIClient` 适配器内部对 openai SDK 的调用，**不用** `mock.patch` 指认内部实现，而是给它一个**可注入的传输层**（`http_client` 参数，第 3 节代码）。这里有个实测教训：最初用 respx 想全局拦截 openai 的请求，结果**失败**——openai SDK 自建 httpx client，拦截层对它不生效，请求直接打到了真网络。把 `http_client` 改成构造参数后，替换回到了"构造函数注入"的正轨。又一次验证：**可测性是设计出来的，不是 patch 出来的**。

```python
# tests/integration/test_llm_contract.py —— OpenAIClient 契约（真 SDK 序列化栈）
import httpx, pytest
from agentutils.adapters import OpenAIClient

REPLY_JSON = {"choices": [{"message": {"content": "周三发布"}}]}

def make_client(handler):
    calls = []
    def wrapped(request):
        calls.append(request)
        return handler(request)
    http = httpx.Client(transport=httpx.MockTransport(wrapped))
    return (OpenAIClient(base_url="https://api.example.com/v1",
                         api_key="test-key", http_client=http), calls)

def test_openai_client_contract():
    client, calls = make_client(lambda req: httpx.Response(200, json=REPLY_JSON))
    assert client.chat("sys", "hi") == "周三发布"
    assert len(calls) == 1
    assert b"gpt-4o-mini" in calls[0].read()        # 模型名进请求体

def test_openai_client_401_raises():
    client, _ = make_client(lambda req: httpx.Response(401, json={"error": "bad"}))
    with pytest.raises(Exception, match="401"):
        client.chat("s", "u")
```

`httpx.MockTransport` 走**真实 SDK 序列化与响应解析**（只换传输），比手写 Fake 更能抓契约偏差；想再进一步就上 07 篇的本地假服务端或 testcontainers。

`.github/workflows/ci.yml`（Python 主体 + C++ 扩展位；若仓库含 C++ 扩展则启用 cpp job——clang-tidy 的接法 08/11 篇已给全）：

## 9. 面试高频问题清单（20 题带要点）

1. **单元测试和集成测试的边界怎么定？** 看真实 I/O 与真实协作；单元层测逻辑分支、集成层测真契约，比例随项目类型（金字塔 vs 奖杯）。
2. **什么是好的单元测试？** FIRST：快/独立/可重复/自校验/及时；红了能直接指向坏的哪个行为。
3. **pytest 的 assert 为什么比 unittest 强？** 导入期 AST 断言重写，输出中间子表达式值；helper 里要 register_assert_rewrite。
4. **fixture 的作用域怎么选？** 默认 function；昂贵只读资源才升 module/session；宽作用域禁放可变状态。
5. **conftest.py 是什么？** 按目录层级自动加载的 pytest 配置/fixture 载体；子目录同名 fixture 覆盖上层，实现分层。
6. **Mock/Stub/Spy/Fake 的区别？** Stub 管输入、Spy/Mock 管交互（事后/事前）、Fake 是有真实行为的轻实现、Dummy 填位。
7. **什么时候不该用 mock？** 不 mock 你不拥有的接口（第三方 SDK）——收进自家端口再造替身；状态型依赖优先 Fake。
8. **patch 常见的坑？** patch 定义处对 from-import 无效（要 patch 使用处）、属性名/模块名写错、值拷贝在导入期固化——三类现场要能复述。
9. **什么是 property-based 测试？** 陈述普适性质（往返/不变量/参考模型等价），引擎生成+缩小反例；适合解析器、纯算法、有状态组件。
10. **TDD 的红绿重构是什么？你的实践节奏？** 测试先行的心智模型；实践中"新代码先行/同期，遗留代码 characterization 先行"。
11. **什么是 seam（接缝）？** Feathers：不编辑现场就能改变行为的点；构造函数注入是最简 object seam，链接期替换是 C/C++ 的 link seam。
12. **接手无测试的遗留代码第一步做什么？** characterization 测试锁现状（值实测），再重构；改行为时先改测试。
13. **测试代码要不要 DRY？** setup 层大胆去重，断言层宁可重复；错误的抽象比重复贵（Sandi Metz）。
14. **flaky 测试怎么治理？** 识别→隔离（quarantine 不阻塞但限期）→归因（sleep/端口/环境泄漏/时区清单）→超期即删；重试必须留记录。
15. **覆盖率 100% 为什么不是目标？** 它是"哪些行从未执行"的诊断指标；无断言测试与分支盲区不被它反映；正解是改动行覆盖门禁。
16. **静态类型检查在测试体系里的位置？** 零成本全量类型层测试；None 流入、Optional 返回这类 bug 的第一道闸；基线+增量 strict 落地。
17. **为什么要 fuzzing？测试用例不够吗？** 示例/性质测试的输入空间仍是人定义的；覆盖引导变异能到达人想不到的深层路径，配 ASan 抓内存 UB。
18. **基准测试为什么忽高忽低？** 频率调节/后台进程/缓存冷热/GC/常量折叠与死代码；纪律是预热、多次 repeat、报 min、同环境对比。
19. **C++ 测试和 Python 测试最大的差异？** 替身机制（链接期 vs 运行时）、内存与 UB 需 Sanitizer 兜底、死亡测试把崩溃变规格；双语言各测各的母语。
20. **你如何把测试文化带进团队？** 门禁自动化（红不可合并）+ 分层反馈（PR 十分钟内）+ flaky 治理条款 + README 测试文档——机制优先于号召。

## 2. 核心模块全量代码

### 2.1 envs.py——向量化环境（本章精华）

```python
"""环境封装: 向量化环境(同步) + 评估环境"""
import numpy as np
import gymnasium as gym

class SyncVecEnv:
    """同步向量化环境。契约(与 Stable-Baselines3 VecEnv 一致):
    step 返回的 obs 中, 已 done 的 env 是"自动 reset 后的新观测";
    真实末状态放在 finals[i](供 truncated 的 bootstrap), 由 has_final 标记。
    terminated 与 truncated 分开返回。"""
    def __init__(self, env_id, n, seed=0):
        self.envs = [gym.make(env_id) for _ in range(n)]
        self.n = n
        self.obs_dim = self.envs[0].observation_space.shape[0]
        self.act_dim = self.envs[0].action_space.n
        self.reset(seed=seed)

    def reset(self, seed=None):
        obs = np.zeros((self.n, self.obs_dim), dtype=np.float32)
        for i, e in enumerate(self.envs):
            o, _ = e.reset(seed=None if seed is None else seed + i)
            obs[i] = o
        return obs

    def step(self, actions):
        obs    = np.zeros((self.n, self.obs_dim), dtype=np.float32)
        rewards = np.zeros(self.n, dtype=np.float32)
        terms  = np.zeros(self.n, dtype=np.float32)   # 真终止
        truns  = np.zeros(self.n, dtype=np.float32)   # 截断
        finals = np.zeros((self.n, self.obs_dim), dtype=np.float32)
        has_final = np.zeros(self.n, dtype=bool)
        for i, (e, a) in enumerate(zip(self.envs, actions)):
            o, r, t, tr, _ = e.step(int(a))
            if t or tr:
                finals[i], has_final[i] = o, True     # 先存真实末状态
                o, _ = e.reset()                      # 自动重置, 新回合开始
            obs[i], rewards[i], terms[i], truns[i] = o, r, t, tr
        return obs, rewards, terms, truns, finals, has_final

def make_eval_env(env_id):
    return gym.make(env_id)
```

两个设计决定，都是"血换来的"：

- **done 后自动 reset 并返回新观测**：这是所有向量库（SB3/rlpyt/tianshou）的统一契约——训练循环不该感知 reset。忘记它是本章 §6 的头号事故。
- **finals 单独传出**：`truncated` 的 bootstrap 需要"真实末状态"，而自动 reset 把它覆盖了——不单独传，截断语义就丢了。

### 2.2 network.py 与 buffer.py

```python
# network.py —— 与 08/09 章同款
import torch
import torch.nn as nn
from torch.distributions import Categorical

class ActorCritic(nn.Module):
    def __init__(self, obs_dim, act_dim, hidden=128):
        super().__init__()
        self.trunk = nn.Sequential(nn.Linear(obs_dim, hidden), nn.ReLU())
        self.pi_head = nn.Linear(hidden, act_dim)
        self.v_head = nn.Linear(hidden, 1)
    def forward(self, x):
        h = self.trunk(x)
        return Categorical(logits=self.pi_head(h)), self.v_head(h)
    def value(self, x):
        return self.v_head(self.trunk(x))
```

```python
# buffer.py —— RolloutBuffer + 跨环境 GAE
import numpy as np

class RolloutBuffer:
    def __init__(self, n_envs):
        self.n = n_envs
        self.reset()
    def reset(self):
        self.obs, self.acts, self.logps = [], [], []
        self.rews, self.terms, self.truns = [], [], []
        self.last_obs = [None] * self.n          # 各 env 末尾的"未来状态"
        self.last_term = [True] * self.n
    def add(self, obs, act, logp, rew, term, trunc, next_obs, finals, has_final):
        self.obs.append(obs); self.acts.append(act)
        self.logps.append(logp); self.rews.append(rew)
        self.terms.append(term); self.truns.append(trunc)
        for i in range(self.n):
            if term[i]:                     # 真终止: 无未来
                self.last_obs[i], self.last_term[i] = None, True
            elif trunc[i]:                  # 截断: 未来 = 真实末状态
                self.last_obs[i], self.last_term[i] = finals[i], False
            else:
                self.last_obs[i], self.last_term[i] = next_obs[i], False
    def __len__(self):
        return len(self.rews)

def compute_gae(vals_TxE, rews_TxE, terms_TxE, next_vals_nE, gamma=0.99, lam=0.95):
    """跨环境 GAE。next_vals: 各 env 缓冲区末尾 bootstrap(真终止=0, 截断=V(真实末状态))"""
    T, n = rews_TxE.shape
    adv = np.zeros((T, n), dtype=np.float32)
    gae = np.zeros(n, dtype=np.float32)
    for t in reversed(range(T)):
        next_v = next_vals_nE.copy()
        if t + 1 < T:
            next_v = vals_TxE[t + 1]
        nonterm = 1.0 - terms_TxE[t]        # 只有真终止清零, 截断保留 bootstrap
        delta = rews_TxE[t] + gamma * next_v * nonterm - vals_TxE[t]
        gae = delta + gamma * lam * gae * nonterm
        adv[t] = gae
    return adv, adv + vals_TxE              # λ-return
```

跨环境 GAE 的要点：每个 env 一条独立的递推链（`gae` 是长度 $n$ 的向量），`done` 只截断**该 env 自己的链**——这是向量化相比单环境循环的实质复杂度所在。

### 2.3 ppo.py——训练器（采集/更新/评估/checkpoint）

```python
import os, time
import numpy as np
import torch
import torch.nn as nn
from .network import ActorCritic
from .buffer import RolloutBuffer, compute_gae
from .envs import SyncVecEnv, make_eval_env

class PPOTrainer:
    def __init__(self, env_id="CartPole-v1", n_envs=8, rollout_steps=256,
                 total_steps=200_000, lr=7e-4, gamma=0.99, lam=0.95,
                 clip=0.2, epochs=10, minibatch=256, vf_coef=0.5,
                 ent_coef=0.01, eval_every=10_000, eval_episodes=10,
                 ckpt_dir="checkpoints", seed=0, device="cpu"):
        self.cfg = dict(env_id=env_id, n_envs=n_envs, rollout_steps=rollout_steps)
        self.env = SyncVecEnv(env_id, n_envs, seed=seed)
        self.eval_env = make_eval_env(env_id)
        self.ac = ActorCritic(self.env.obs_dim, self.env.act_dim).to(device)
        self.opt = torch.optim.Adam(self.ac.parameters(), lr=lr)
        self.buf = RolloutBuffer(n_envs)
        self.gamma, self.lam, self.clip = gamma, lam, clip
        self.epochs, self.minibatch = epochs, minibatch
        self.vf_coef, self.ent_coef = vf_coef, ent_coef
        self.total_steps, self.eval_every = total_steps, eval_every
        self.eval_episodes, self.ckpt_dir = eval_episodes, ckpt_dir
        self.device = device
        os.makedirs(ckpt_dir, exist_ok=True)
        self.rng = np.random.default_rng(seed)          # 全部随机源统一受 seed 管

    def evaluate(self):
        """固定评估协议: 贪心 + 独立种子 + 10 回合均值(全系列的评估纪律)"""
        total = 0.0
        for ep in range(self.eval_episodes):
            obs, _ = self.eval_env.reset(seed=999_000 + ep)
            done = False
            while not done:
                with torch.no_grad():
                    dist, _ = self.ac(torch.tensor(obs, dtype=torch.float32,
                                                   device=self.device))
                    a = int(torch.argmax(dist.logits))
                obs, r, term, trunc, _ = self.eval_env.step(a)
                total += r; done = term or trunc
        return total / self.eval_episodes

    def checkpoint(self, step, extra=None):
        path = os.path.join(self.ckpt_dir, f"ppo_{step:07d}.pt")
        torch.save({"model": self.ac.state_dict(),        # 模型与优化器分开存,
                    "opt": self.opt.state_dict(),         # 恢复训练才完整
                    "step": step, "extra": extra or {}}, path)
        return path

    def collect(self, step_count):
        """采 rollout_steps × n_envs 步, 返回 (新步数, 本批完成的回合回报列表)"""
        if not hasattr(self, "last_obs"):
            self.last_obs = self.env.reset()
        obs = self.last_obs
        eps_returns, cur_ret = [], np.zeros(self.cfg["n_envs"], dtype=np.float32)
        target = step_count + self.cfg["rollout_steps"] * self.cfg["n_envs"]
        while step_count < target:
            with torch.no_grad():
                dist, _ = self.ac(torch.tensor(obs, device=self.device))
                a = dist.sample()
                logp = dist.log_prob(a)
            nobs, r, term, trunc, finals, has_final = self.env.step(a.cpu().numpy())
            self.buf.add(obs, a.cpu().numpy(), logp.cpu().numpy(), r,
                         term, trunc, nobs, finals, has_final)
            cur_ret += r
            for i in range(self.cfg["n_envs"]):
                if term[i] or trunc[i]:
                    eps_returns.append(float(cur_ret[i])); cur_ret[i] = 0.0
            obs = nobs
            step_count += self.cfg["n_envs"]
        self.last_obs = obs
        return step_count, eps_returns

    def update(self):
        b, n = self.buf, self.cfg["n_envs"]
        T = len(b.rews)
        obs_Tn  = np.array(b.obs)              # (T, n, obs_dim)
        acts_Tn = np.array(b.acts)             # (T, n)
        logps_Tn= np.array(b.logps)            # (T, n)
        rews_Tn = np.array(b.rews)             # (T, n)
        terms_Tn= np.array(b.terms)
        next_vals = np.zeros(n, dtype=np.float32)   # 各 env 的末尾 bootstrap
        with torch.no_grad():
            for i in range(n):
                if b.last_obs[i] is not None:
                    next_vals[i] = self.ac.value(torch.tensor(
                        b.last_obs[i], dtype=torch.float32,
                        device=self.device)).item()
            vals = np.zeros((T, n), dtype=np.float32)
            for t in range(T):
                vals[t] = self.ac.value(torch.tensor(
                    obs_Tn[t], device=self.device)).squeeze(-1).cpu().numpy()
        adv, ret = compute_gae(vals, rews_Tn, terms_Tn, next_vals, self.gamma, self.lam)
        B = T * n                              # 摊平: 跨 env 联合 batch
        S   = torch.tensor(obs_Tn.reshape(B, -1), device=self.device)
        A   = torch.tensor(acts_Tn.reshape(B), device=self.device)
        LP  = torch.tensor(logps_Tn.reshape(B), dtype=torch.float32, device=self.device)
        ADV = torch.tensor(adv.reshape(B), dtype=torch.float32, device=self.device)
        RET = torch.tensor(ret.reshape(B), dtype=torch.float32, device=self.device)
        ADV = (ADV - ADV.mean()) / (ADV.std() + 1e-8)   # 全批归一化一次(09章口径)
        idx = np.arange(B)
        for _ in range(self.epochs):
            self.rng.shuffle(idx)
            for s in range(0, B, self.minibatch):
                mb = idx[s:s + self.minibatch]
                dist, v = self.ac(S[mb])
                logp = dist.log_prob(A[mb])
                ratio = (logp - LP[mb]).exp()
                surr1 = ratio * ADV[mb]
                surr2 = torch.clamp(ratio, 1 - self.clip, 1 + self.clip) * ADV[mb]
                pg = -torch.min(surr1, surr2).mean()
                vl = nn.functional.smooth_l1_loss(v.squeeze(-1), RET[mb])
                loss = pg + self.vf_coef * vl - self.ent_coef * dist.entropy().mean()
                self.opt.zero_grad(); loss.backward()
                nn.utils.clip_grad_norm_(self.ac.parameters(), 0.5)
                self.opt.step()
        self.buf.reset()
        return B

    def train(self):
        step_count, ep_returns, eval_hist = 0, [], []
        t0 = time.time(); next_eval = self.eval_every
        while step_count < self.total_steps:
            step_count, eps = self.collect(step_count)
            ep_returns += eps
            self.update()
            if step_count >= next_eval:
                score = self.evaluate()
                eval_hist.append((step_count, score))
                p = self.checkpoint(step_count, {"eval": score})
                print(f"step {step_count:7d} | eval={score:6.1f} | "
                      f"train50={np.mean(ep_returns[-50:]) if ep_returns else float('nan'):6.1f} "
                      f"| {time.time()-t0:5.1f}s | {p}")
                next_eval += self.eval_every
        return eval_hist
```

## 4. gymnasium 自定义环境规范

RL 工程里一半的 bug 源于自定义环境不符合 API 契约。最小规范模板（含全部边界情况）：

```python
import gymnasium as gym
from gymnasium import spaces
import numpy as np

class MyEnv(gym.Env):
    metadata = {"render_modes": []}

    def __init__(self, max_steps=500):
        super().__init__()
        self.observation_space = spaces.Box(low=-np.inf, high=np.inf,
                                            shape=(4,), dtype=np.float32)
        self.action_space = spaces.Discrete(2)
        self.max_steps = max_steps
        self._np_random = None            # 由 reset(seed=) 注入

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)          # 必须: 接管 self.np_random
        self.state = self.np_random.uniform(-0.05, 0.05, size=4)
        self.t = 0
        return self.state.astype(np.float32), {}   # (obs, info)

    def step(self, action):
        # ① 推进状态(用 self.np_random, 不要用全局 np.random!)
        self.state = self.state + 0.01 * (2 * action - 1) \
                     + self.np_random.normal(0, 0.01, size=4)
        self.t += 1
        # ② 奖励
        reward = 1.0
        # ③ 双终止语义, 分开判定
        x, _, th, _ = self.state
        terminated = bool(abs(x) > 2.4 or abs(th) > 0.2095)   # 真终止
        truncated  = bool(self.t >= self.max_steps)           # 时间截断
        # ④ obs 必须 float32 且在 space 内
        obs = np.clip(self.state, -10, 10).astype(np.float32)
        return obs, reward, terminated, truncated, {"step": self.t}
```

契约清单（每条都对应一类真实 bug）：

1. **随机源唯一**：一切随机用 `self.np_random`（`super().reset(seed=)` 注入），混入 `np.random`/`random` 会让 `env.reset(seed=s)` 的可复现性形同虚设；
2. **terminated/truncated 语义分明**：真终止（任务失败/达成）vs 截断（时间上限）——01 章埋线、05/06 章事故、本章 buffer 的 `finals` 机制，全靠这两个标志的诚实；
3. **done 后 step 是未定义行为**：不要假设用户不会多调 step；向量化封装（§2.1）负责自动 reset；
4. **obs 的 dtype/space 一致**：声明 `float32` 就返回 `float32`，声明的上下界要真实（network 输入非法值会静默污染训练）；
5. **reset 必须完全重置内部状态**（包括计数器、缓存、历史），隐藏的"上回合残留"是"换个 seed 就复现不出来"类 bug 的头号来源；
6. **奖励不依赖未定义状态**：终止那一步的奖励要在 `terminated=True` 的同一 step 里给出，不要推迟。

## 5. 不收敛 10 大原因排查表

按出现频率排序（"经验做法"，来自本系列全部实验 + 社区高频问题；先查 1–5 再查 6–10）：

| # | 症状 | 原因 | 检查/修复 |
| --- | --- | --- | --- |
| 1 | loss 下降但 eval 恒定 | target 计算没包 `no_grad`（06 章）或 `logp_old` 用当前网络重算（09 章） | 检查 ratio 是否恒为 1 |
| 2 | 前期学得动、中期突然归零 | `truncated` 被当 `terminated` 清零 bootstrap（05/06 章事故） | 打印两种 done 的比例；buffer 存分开的标志 |
| 3 | 梯度范数巨大、eval 卡随机水平 | 共享主干 + MSE 价值损失量级失控（08 章实测事故） | 换 Huber；查 v_loss 与 pg_loss 的量级比 |
| 4 | 完全不学、优势全零 | GAE 递推链被 done 错误截断 / 向量化环境没自动 reset（本章 §6 事故） | gymnasium 警告 "step after terminated" 必看 |
| 5 | 学到一半震荡加剧、发散 | lr 过大 + epochs 过多；KL 飙升（09 章消融） | 监控 approx_kl（健康 0.003–0.03）；降 lr |
| 6 | 回报上去了但行为明显错 | reward hacking / 奖励写错（10 章）：奖励与真实目标错位 | 人工看 10 条轨迹；检查"刷分路径" |
| 7 | 每次跑结果完全不同 | 种子没固定全：env.reset / np / torch / python random 四处 | 统一 seed 注入（本章 `self.rng` 模式） |
| 8 | 表格法不收敛（03–05 章） | 终止状态被更新 / γ=1 且策略可能死循环 / θ 太小 | 终止状态 continue；continuing 用 γ<1 |
| 9 | 离线设定下学出胡乱策略 | 外推误差：对未见动作乐观（11 章实测） | 策略约束/悲观估计；检查数据覆盖度 |
| 10 | "收敛"到次优策略就躺平 | 熵塌缩探索死亡（08 章误区 5）或 ε 衰减过早（05 章） | 监控策略熵/ε 曲线；提 ent_coef |

通用诊断顺序（经验做法）：**固定种子重跑 → 看梯度范数与 loss 量级 → 打印 5 条完整轨迹人工检查 → 二分法砍组件**（去掉熵项/归一化/价值损失各试一次，定位引入坏处的改动）。

## 6. 一次真实的调试过程（本章项目开发实录）

本章 pipeline 第一版跑起来后 eval 卡死 17.2、train50=0.0。完整调试轨迹（教学价值在于"症状→定位"的推理链）：

1. **症状**：每 10k 步 eval 恒等于 17.2（分毫不差），train50 全 0。
2. **线索一**：train50 = 0.0 不可能——CartPole 每步至少 +1，只有"回合回报从未被记录"才会是 0。怀疑回合判定链路。
3. **线索二**：控制台里躺着 gymnasium 的警告——`You are calling 'step()' even though this environment has already returned terminated = True`。**终止后没有 reset，还在对同一环境继续 step**（未定义行为，状态停格，所以 eval 恒定）。
4. **定位**：`SyncVecEnv.step` 只是转发 step，没有任何 done 后 reset 逻辑；`collect` 也没有。
5. **修复**：step 契约改为"done 后自动 reset + finals 传出真实末状态"（§2.1 现版）；同步修 buffer 的 `last_obs` 三分支（term/trunc/进行中）。
6. **验证**：重跑 100k 步 → eval 10k 步到 483、61k 步满分（§3 结果）。
7. **附带收获**：修复后还发现 train50 后期下降的"假象"（§3 读数 2），顺带把"训练统计滞后"写进了方法论。

**教训**：RL 的静默失败远多于报错失败——`terminated` 后继续 step 不抛异常、obs 停格不告警、advantage 全零 loss 照样下降。**永远读环境库的 warning，永远打印一批数据人眼过一遍**。

## 7. 面试高频 25 题（带要点）

**基础概念**

1. RL 与监督学习的本质区别？→ 数据分布由策略决定/奖励延迟/无正确答案只有分数（01）。
2. MDP 五元组与马尔可夫性？→ $(S,A,P,R,\gamma)$；历史无关；不满足时帧叠加/RNN（01）。
3. 折扣因子的两重身份？→ 收敛保证 + 视野旋钮（视野 ≈ $1/(1-\gamma)$）（01）。
4. episodic vs continuing，terminated vs truncated？→ 后者是工程口径：截断不清零 bootstrap（01/05）。
5.贝尔曼期望方程与最优方程的差别？→ 策略平均 vs max；线性 vs 非线性（03）。

**表格法**

6. MC 与 TD 的偏差-方差？→ MC 无偏高方差、TD 有偏差低方差；n-step/GAE 参数化光谱（04/08）。
7. on-policy vs off-policy？→ 两问：数据谁采、更新谁；Q-learning off（可复用数据），SARSA on（学含探索的值）（05）。
8. Q-learning 为什么会高估？→ max 作用在噪声估计上；Double DQN 解耦选动作与评估（05/06）。
9. CliffWalking 上 Q-learning 与 SARSA 的路径差异？→ 最短贴崖 vs 安全离崖；"学得优"vs"走得稳"（05）。
10. 表格法什么时候够用？→ 状态可枚举、能覆盖、表示稳定；否则函数近似（05）。

**深度 RL**

11. Deadly Triad？→ 自举+函数近似+off-policy；正反馈高估回路（04/06）。
12. 经验回放的两个作用？→ 打破时间相关性 + 样本复用；代价：off-policy 偏差（06）。
13. 目标网络为什么稳定？→ 冻结自举目标，"追自己尾巴"→"朝固定靶回归"；偏差换稳定（06）。
14. DQN 的 loss 该用 MSE 还是 Huber？→ Huber 对离群 TD 误差钝感；梯度裁剪配合（06）。
15. 为什么 DQN 处理不了连续动作？→ 内层 max 是优化问题；策略梯度直接输出分布参数（06/07）。

**策略梯度**

16. 策略梯度定理怎么来的？→ log-导数技巧消环境因子；score function × 回报加权（07）。
17. 基线为什么无偏降方差？→ $\mathbb{E}[\nabla\log\pi \cdot b]=0$（$\sum_a\pi\nabla\log\pi=0$）；最优基线是 score 平方加权均值（07）。
18. 优势函数与 TD 误差的关系？→ $\mathbb{E}[\delta_t]=A(s,a)$；一个 V 网络三用（基线/自举/优势估计）（08）。
19. GAE 的 λ 是什么？→ k 步优势的指数加权；λ=0 纯 TD、λ=1 MC；实测 λ=0 不学习（08）。
20. PPO 的 clip 为什么稳定？→ 挡"继续推"不挡"退回"；数据有效域内截断优化；无 clip 消融 KL 飙 3.2（09）。
21. PPO 调参看什么指标？→ approx_kl（0.003–0.03）与 clipfrac（<0.3）；loss 几乎无信息（09）。

**LLM 对齐**

22. RLHF 三阶段？→ SFT → RM（Bradley-Terry MLE）→ PPO+KL；InstructGPT 1.3B 胜 175B（10）。
23. KL 惩罚为什么必须？→ RM 只在数据分布附近可信；β 的 U 型权衡（实测 −100 → +12.5 → 10.5）（10）。
24. DPO 推导主线？→ KL 约束 RL 闭式解 → 隐式奖励 $r=\beta\log(\pi/\pi_{ref})$ → 代回 BT 时 Z(x) 相消 → sigmoid 交叉熵（10）。
25. GRPO 为什么敢删 Critic？→ 组均值 = 免费的 prompt 级基线（基线只需与动作无关）；代价是推理 ×G（10）。

答题要领：**每题先一句定义、再一句机制、最后给一个本系列的实验锚点**（如"λ=0 实测不学习"、"无 clip KL 飙到 3.2"）——有实验锚点的回答在面试里辨识度极高。

## 8. 常见误区

1. **把 train 回报当训练进度**（§3 读数 2 的实证：eval 500 时 train50 才 111）。评估协议（独立种子 + 贪心 + 多回合均值）是唯一可信进度条。
2. **向量化环境不自动 reset**（§6 实录）。gymnasium 的 warning 是唯一线索，别忽略。
3. **checkpoint 只存模型不存优化器**。恢复训练时 Adam 的动量/二阶矩全丢，等效学习率突变。正确姿势：`{"model", "opt", "step"}` 三件套（§2.3）。
4. **自定义环境里用全局随机源**。`np.random` 绕过了 `reset(seed=)` 的注入，种子固定形同虚设（§4 清单 1）。
5. **GAE 的 bootstrap 在向量化和截断下想当然**。跨 env 递推链 + finals 语义（§2.2/2.3）是向量化实现里最容易错的 20 行——写完用"小 rollout 手算对照"验证一次。

## 本篇小结

- 完整 pipeline 的分层：环境封装（契约严格）→ 缓冲与 GAE（每 env 独立递推链）→ 训练器（采集/更新/评估/checkpoint 解耦）→ 入口脚本；全部 CPU，100k 步 12.1 秒收敛到 eval 500。
- 向量化环境的契约：done 后自动 reset、finals 传出真实末状态、terminated/truncated 分开——三条都对应真实 bug。
- 自定义环境六条契约：随机源唯一、双终止语义、done 后行为、obs dtype/space 一致、reset 完全重置、奖励即时性。
- 不收敛排查是**顺序工程**：10 大原因表按频率排序，配合"固定种子→梯度范数→人看轨迹→二分砍组件"的诊断流。
- 面试 25 题的答法：定义 → 机制 → 实验锚点；本系列的每个实验数字都是你的弹药库。
